# Módulo de Gastos Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Agregar un módulo backend-persistido de Gastos (categorías, gastos puntuales, gastos recurrentes con generación automática) bajo Tesorería, sincronizado como líneas de solo lectura en Flujo de Caja.

**Architecture:** Nuevo package backend `cl.slimerp.gastos` (tres tablas: `categoria_gasto`, `gasto_recurrente`, `gasto`), siguiendo el patrón CRUD ya usado por `catalogo.Categoria` y `compras.Compra`; un `@Scheduled` job genera instancias de `Gasto` a partir de `GastoRecurrente`. Nuevo feature Angular `features/gastos` con tres pestañas (Categorías, Recurrentes, Gastos). La sincronización a Flujo de Caja replica exactamente el patrón ya implementado para Cobros (`CobrosSyncService` → `SyncedIncome`), agregando `GastosSyncService` → `SyncedExpense`.

**Tech Stack:** Spring Boot 3 / Java 21, JPA + Flyway (Postgres), Angular standalone components + Angular Material + RxJS (template-driven forms).

**Spec:** `docs/superpowers/specs/2026-09-11-modulo-gastos-design.md`

package cl.slimerp.gastos;

import cl.slimerp.common.PaginaResponse;
import cl.slimerp.config.TenantContext;
import jakarta.validation.Valid;
import org.springframework.data.domain.PageRequest;
import org.springframework.data.domain.Sort;
import org.springframework.http.ResponseEntity;
import org.springframework.security.access.prepost.PreAuthorize;
import org.springframework.web.bind.annotation.*;

import java.util.List;

@RestController
@RequestMapping("/api/gastos/categorias")
public class CategoriaGastoController {

    private final CategoriaGastoRepository categoriaGastoRepository;

    public CategoriaGastoController(CategoriaGastoRepository categoriaGastoRepository) {
        this.categoriaGastoRepository = categoriaGastoRepository;
    }

    @GetMapping
    @PreAuthorize("hasAuthority('GASTOS_VER')")
    public List<CategoriaGasto> listar() {
        return categoriaGastoRepository.findByTenantIdAndActivoTrue(TenantContext.getTenantId());
    }

    @GetMapping("/pagina")
    @PreAuthorize("hasAuthority('GASTOS_VER')")
    public PaginaResponse<CategoriaGasto> listarPagina(
            @RequestParam(required = false) String q,
            @RequestParam(defaultValue = "0") int pagina,
            @RequestParam(defaultValue = "10") int tamano) {
        String busqueda = "%" + (q == null ? "" : q.trim().toLowerCase()) + "%";
        var pageable = PageRequest.of(pagina, tamano, Sort.by("nombre").ascending());
        return PaginaResponse.de(categoriaGastoRepository.buscar(TenantContext.getTenantId(), busqueda, pageable));
    }

    @PostMapping
    @PreAuthorize("hasAuthority('GASTOS_EDITAR')")
    public ResponseEntity<CategoriaGasto> crear(@Valid @RequestBody CategoriaGastoRequest request) {
        CategoriaGasto categoria = CategoriaGasto.builder()
                .tenantId(TenantContext.getTenantId())
                .nombre(request.nombre())
                .build();
        return ResponseEntity.ok(categoriaGastoRepository.save(categoria));
    }

    @PutMapping("/{id}")
    @PreAuthorize("hasAuthority('GASTOS_EDITAR')")
    public ResponseEntity<CategoriaGasto> actualizar(@PathVariable Long id, @Valid @RequestBody CategoriaGastoRequest request) {
        return categoriaGastoRepository.findByIdAndTenantIdAndActivoTrue(id, TenantContext.getTenantId())
                .map(categoria -> {
                    categoria.setNombre(request.nombre());
                    return ResponseEntity.ok(categoriaGastoRepository.save(categoria));
                })
                .orElseGet(() -> ResponseEntity.notFound().build());
    }

    @DeleteMapping("/{id}")
    @PreAuthorize("hasAuthority('GASTOS_EDITAR')")
    public ResponseEntity<Void> eliminar(@PathVariable Long id) {
        return categoriaGastoRepository.findByIdAndTenantIdAndActivoTrue(id, TenantContext.getTenantId())
                .map(categoria -> {
                    categoria.setActivo(false);
                    categoriaGastoRepository.save(categoria);
                    return ResponseEntity.noContent().<Void>build();
                })
                .orElseGet(() -> ResponseEntity.notFound().build());
    }
}

package cl.slimerp.gastos;

import cl.slimerp.config.TenantContext;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

@Service
public class GastoRecurrenteService {

    private final GastoRecurrenteRepository gastoRecurrenteRepository;
    private final CategoriaGastoRepository categoriaGastoRepository;

    public GastoRecurrenteService(GastoRecurrenteRepository gastoRecurrenteRepository,
                                   CategoriaGastoRepository categoriaGastoRepository) {
        this.gastoRecurrenteRepository = gastoRecurrenteRepository;
        this.categoriaGastoRepository = categoriaGastoRepository;
    }

    @Transactional
    public GastoRecurrente crear(GastoRecurrenteRequest request) {
        Long tenantId = TenantContext.getTenantId();
        validarCategoria(request.categoriaGastoId(), tenantId);
        GastoRecurrente plantilla = GastoRecurrente.builder()
                .tenantId(tenantId)
                .categoriaGastoId(request.categoriaGastoId())
                .monto(request.monto())
                .descripcion(request.descripcion())
                .diaMes(request.diaMes())
                .fechaInicio(request.fechaInicio())
                .fechaFin(request.fechaFin())
                .build();
        return gastoRecurrenteRepository.save(plantilla);
    }

    @Transactional
    public GastoRecurrente actualizar(Long id, GastoRecurrenteRequest request) {
        Long tenantId = TenantContext.getTenantId();
        GastoRecurrente plantilla = gastoRecurrenteRepository.findByIdAndTenantIdAndActivoTrue(id, tenantId)
                .orElseThrow(() -> new IllegalArgumentException("Gasto recurrente no encontrado: " + id));
        validarCategoria(request.categoriaGastoId(), tenantId);
        plantilla.setCategoriaGastoId(request.categoriaGastoId());
        plantilla.setMonto(request.monto());
        plantilla.setDescripcion(request.descripcion());
        plantilla.setDiaMes(request.diaMes());
        plantilla.setFechaInicio(request.fechaInicio());
        plantilla.setFechaFin(request.fechaFin());
        return gastoRecurrenteRepository.save(plantilla);
    }

    @Transactional
    public void desactivar(Long id) {
        Long tenantId = TenantContext.getTenantId();
        GastoRecurrente plantilla = gastoRecurrenteRepository.findByIdAndTenantIdAndActivoTrue(id, tenantId)
                .orElseThrow(() -> new IllegalArgumentException("Gasto recurrente no encontrado: " + id));
        plantilla.setActivo(false);
        gastoRecurrenteRepository.save(plantilla);
    }

    private void validarCategoria(Long categoriaGastoId, Long tenantId) {
        categoriaGastoRepository.findByIdAndTenantIdAndActivoTrue(categoriaGastoId, tenantId)
                .orElseThrow(() -> new IllegalArgumentException("Categoría de gasto no encontrada: " + categoriaGastoId));
    }
}

package cl.slimerp.gastos;

import cl.slimerp.config.TenantContext;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

@Service
public class GastoService {

    private final GastoRepository gastoRepository;
    private final CategoriaGastoRepository categoriaGastoRepository;

    public GastoService(GastoRepository gastoRepository, CategoriaGastoRepository categoriaGastoRepository) {
        this.gastoRepository = gastoRepository;
        this.categoriaGastoRepository = categoriaGastoRepository;
    }

    @Transactional
    public Gasto crear(GastoRequest request) {
        Long tenantId = TenantContext.getTenantId();
        validarCategoria(request.categoriaGastoId(), tenantId);
        Gasto gasto = Gasto.builder()
                .tenantId(tenantId)
                .categoriaGastoId(request.categoriaGastoId())
                .monto(request.monto())
                .descripcion(request.descripcion())
                .fecha(request.fecha())
                .build();
        return gastoRepository.save(gasto);
    }

    @Transactional
    public Gasto actualizar(Long id, GastoRequest request) {
        Long tenantId = TenantContext.getTenantId();
        Gasto gasto = gastoRepository.findByIdAndTenantIdAndActivoTrue(id, tenantId)
                .orElseThrow(() -> new IllegalArgumentException("Gasto no encontrado: " + id));
        validarCategoria(request.categoriaGastoId(), tenantId);
        gasto.setCategoriaGastoId(request.categoriaGastoId());
        gasto.setMonto(request.monto());
        gasto.setDescripcion(request.descripcion());
        gasto.setFecha(request.fecha());
        return gastoRepository.save(gasto);
    }

    @Transactional
    public void eliminar(Long id) {
        Long tenantId = TenantContext.getTenantId();
        Gasto gasto = gastoRepository.findByIdAndTenantIdAndActivoTrue(id, tenantId)
                .orElseThrow(() -> new IllegalArgumentException("Gasto no encontrado: " + id));
        gasto.setActivo(false);
        gastoRepository.save(gasto);
    }

    private void validarCategoria(Long categoriaGastoId, Long tenantId) {
        categoriaGastoRepository.findByIdAndTenantIdAndActivoTrue(categoriaGastoId, tenantId)
                .orElseThrow(() -> new IllegalArgumentException("Categoría de gasto no encontrada: " + categoriaGastoId));
    }

package cl.slimerp.gastos;

import cl.slimerp.tenant.Tenant;
import cl.slimerp.tenant.TenantRepository;
import org.springframework.scheduling.annotation.Scheduled;
import org.springframework.stereotype.Component;
import org.springframework.transaction.annotation.Transactional;

import java.time.LocalDate;
import java.time.YearMonth;

/**
 * Genera automáticamente la instancia mensual de cada {@link GastoRecurrente} activo cuyo
 * día del mes calza con hoy. Corre fuera de un hilo de request: no usa {@code TenantContext},
 * itera explícitamente todos los tenants.
 */
@Component
public class GastoRecurrenteGeneratorJob {

    private final TenantRepository tenantRepository;
    private final GastoRecurrenteRepository gastoRecurrenteRepository;
    private final GastoRepository gastoRepository;

    public GastoRecurrenteGeneratorJob(TenantRepository tenantRepository,
                                        GastoRecurrenteRepository gastoRecurrenteRepository,
                                        GastoRepository gastoRepository) {
        this.tenantRepository = tenantRepository;
        this.gastoRecurrenteRepository = gastoRecurrenteRepository;
        this.gastoRepository = gastoRepository;
    }

    @Scheduled(cron = "0 0 3 * * *")
    public void generarInstanciasDelDia() {
        LocalDate hoy = LocalDate.now();
        for (Tenant tenant : tenantRepository.findAll()) {
            generarParaTenant(tenant.getId(), hoy);
        }
    }

    @Transactional
    void generarParaTenant(Long tenantId, LocalDate hoy) {
        for (GastoRecurrente plantilla : gastoRecurrenteRepository.findByTenantIdAndActivoTrue(tenantId)) {
            if (corresponde(plantilla, hoy)) {
                generarInstancia(plantilla, hoy);
            }
        }
    }

    private boolean corresponde(GastoRecurrente plantilla, LocalDate hoy) {
        if (!plantilla.getDiaMes().equals(hoy.getDayOfMonth())) return false;
        if (hoy.isBefore(plantilla.getFechaInicio())) return false;
        return plantilla.getFechaFin() == null || !hoy.isAfter(plantilla.getFechaFin());
    }

    private void generarInstancia(GastoRecurrente plantilla, LocalDate hoy) {
        YearMonth mes = YearMonth.from(hoy);
        // Idempotencia: si el job corre dos veces el mismo día (o se reinicia), no duplica.
        boolean yaExiste = gastoRepository.existsByGastoRecurrenteIdAndFechaBetween(
                plantilla.getId(), mes.atDay(1), mes.atEndOfMonth());
        if (yaExiste) return;

        Gasto gasto = Gasto.builder()
                .tenantId(plantilla.getTenantId())
                .categoriaGastoId(plantilla.getCategoriaGastoId())
                .gastoRecurrenteId(plantilla.getId())
                .monto(plantilla.getMonto())
                .descripcion(plantilla.getDescripcion())
                .fecha(hoy)
                .build();
        gastoRepository.save(gasto);
    }
}

  }

  actualizar(id: number, request: GastoRequest): Observable<Gasto> {

  columnas = ['nombre', 'acciones'];

  filtro = '';
  pagina = 0;
  tamano = 10;

  nombre = '';
  editandoId: number | null = null;
  guardando = false;

  private readonly busqueda$ = new Subject<string>();

  constructor(
    private categoriaGastoService: CategoriaGastoService,
    public auth: AuthService,
  ) {}

  ngOnInit(): void {
    this.busqueda$.pipe(debounceTime(300), distinctUntilChanged()).subscribe(() => {
      this.pagina = 0;
      this.cargar();
    });
    this.cargar();
  }

  ngOnDestroy(): void {
    this.busqueda$.complete();
  }

  cargar(): void {
    this.categoriaGastoService.listarPagina(this.filtro, this.pagina, this.tamano).subscribe((resp) => {
      if (!resp.contenido.length && this.pagina > 0) {
        this.pagina = Math.max(0, this.pagina - 1);
        this.cargar();
        return;
      }
      this.categorias = resp.contenido;
      this.total = resp.total;
    });
  }

  onFiltroChange(): void {
    this.busqueda$.next(this.filtro);
  }

  onPageChange(event: PageEvent): void {
    this.pagina = event.pageIndex;
    this.tamano = event.pageSize;
    this.cargar();
  }

  editar(categoria: CategoriaGasto): void {
    this.editandoId = categoria.id;
    this.nombre = categoria.nombre;
  }

  cancelarEdicion(): void {
    this.editandoId = null;
    this.formulario?.resetForm({ nombre: '' });
  }

  guardar(): void {
    if (!this.nombre.trim()) return;
    const request: CategoriaGastoRequest = { nombre: this.nombre.trim() };
    this.guardando = true;
    mostrarCargando(this.editandoId ? 'Guardando cambios' : 'Creando categoría');
    const obs = this.editandoId
      ? this.categoriaGastoService.actualizar(this.editandoId, request)
      : this.categoriaGastoService.crear(request);
    obs.subscribe({
      next: () => {
        cerrarCargando();
        this.error = '';
        this.guardando = false;
        this.editandoId = null;
        this.formulario?.resetForm({ nombre: '' });
        this.cargar();
      },
      error: (err) => {
        cerrarCargando();
        this.guardando = false;
        this.error = err?.error?.error ?? 'Ocurrió un error. Intenta nuevamente.';
      },
    });
  }

  eliminar(categoria: CategoriaGasto): void {
    this.categoriaGastoService.eliminar(categoria.id).subscribe({
      next: () => {
        this.error = '';
        if (this.editandoId === categoria.id) this.cancelarEdicion();
        this.cargar();
      },
      error: (err) => {
        this.error = err?.error?.error ?? 'Ocurrió un error. Intenta nuevamente.';
      },
    });
  }
}

import { Injectable, inject } from '@angular/core';
import { Observable, firstValueFrom, from } from 'rxjs';
import { GastoService } from '../../../core/services/gasto.service';
import { Gasto } from '../../../core/models/models';
import { mapGastoToSyncedExpense, SyncedExpense } from './synced-expense';

const TAMANO_PAGINA = 200;

@Injectable({ providedIn: 'root' })
export class GastosSyncService {
  private readonly gastoService = inject(GastoService);

  sincronizar(): Observable<SyncedExpense[]> {
    return from(this.cargarTodo());
  }

  private async cargarTodo(): Promise<SyncedExpense[]> {
    const gastos = await this.traerTodos();
    return gastos.map(mapGastoToSyncedExpense);
  }

  /** Pagina hasta traer todos los gastos: `projection.ts` necesita el conjunto completo, no una página. */
  private async traerTodos(): Promise<Gasto[]> {
    const acumulado: Gasto[] = [];
    let pagina = 0;
    for (;;) {
      const resp = await firstValueFrom(this.gastoService.listar({ pagina, tamano: TAMANO_PAGINA }));
      acumulado.push(...resp.contenido);
      if (acumulado.length >= resp.total || resp.contenido.length === 0) break;
      pagina++;
    }
    return acumulado;
  }
}

export function projectMonths(
  state: CashflowState,
  months: readonly MonthKey[],
  openingBalance: number,
  syncedIncomes: readonly SyncedIncome[] = [],
  syncedExpenses: readonly SyncedExpense[] = [],
): MonthProjection[] {
  const result: MonthProjection[] = [];
  let balance = openingBalance;
  let taxProvision = 0;
  for (const month of months) {
    const projection = projectMonth(state, month, balance, taxProvision, syncedIncomes, syncedExpenses);
    result.push(projection);
    balance = projection.closingBalance;
    taxProvision = projection.accumulatedTaxProvision;
  }
  return result;
}

export function projectThrough(
  state: CashflowState,
  throughYear: number,
  syncedIncomes: readonly SyncedIncome[] = [],
  syncedExpenses: readonly SyncedExpense[] = [],
): MonthProjection[] {
  const { baseYear, openingBalance } = state.settings;
  const years = Math.max(1, throughYear - baseYear + 1);
  const months = monthKeysFrom(januaryOf(baseYear), years * MONTHS_PER_YEAR);
  return projectMonths(state, months, openingBalance, syncedIncomes, syncedExpenses);
}

export function projectYear(
  state: CashflowState,
  year: number,
  syncedIncomes: readonly SyncedIncome[] = [],
  syncedExpenses: readonly SyncedExpense[] = [],
): MonthProjection[] {
  return projectThrough(state, year, syncedIncomes, syncedExpenses).slice(-MONTHS_PER_YEAR);
}
